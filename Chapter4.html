      변수의 유효 범위, 스코프

      스코프 : 선언한 변수의 적용 범위
      지역 스코프 : 특정 영역에서만 사용 가능
      ex) 함수 스코프, 블록 스코프
      함수 스코프는 함수 내부에서 선언되어 함수 내부에서만 접근 가능
      var,let,const 동일
      예문) function sum(a,b){ var result = a + b;}
      console.log(result) // 에러 발생
      블록 스코프는 {}, if, for문 블록 내부에서 선언된 변수
      근데 var은 내부에서 선언해도 외부에서도 접근 사용 가능
      (let, const은 변수가 선언 된 블록 안에서만 유효)
      예문) const factor = 5;//큰 외부에 있어 어디든 사용 가능
      function calc(){
      return num * factor;
      }
      {
      const num = 10;//블록 스코프에서만 쓸 수 있음
      //calc함수를 이 블록 안에서 선언했으면 사용 가능했겠지만 그렇지 않으므로
      //이 메서드를 실행하고 싶다면 calc(num) 으로 인자를 받으면 될 듯함.
      let result = calc();
      //확인 결과 됨.
      document.write(`result : ${result}`);
      } 전역 스코프 : 시작 부분에서 변수 선언하여 프로그램 전체에서 사용할 수 있는 스코프
      함수 어디에서나 값을 변경할 수 있음.
      var hi = "hello";
      function change(){ hi = "bye";}
      console.log(hi);//전역 변수인 hello 출력
      change();//변수값을 bye로 변환 전역에서 적용이 됨
      console.log(hi);//적용되어 bye 출력
      함수에서 변수를 선언할 때 var 예약어를 붙이지 않으면 js는 전역 변수로 인식
      function greeting(){hi = "hello";}
      greeting();//함수 실행 시 hi에 hello 할당
      console.log(hi)//전역이라 hello를 출력 가능

      실습1. 입력값을 표시하는 콜백함수 만들기
      function showData(name,age){
      alert(`${name}님의 나이는 ${age}살입니다.`);
      }
      function getData(callBack){//보통 콜백함수의 매개변수는 callback으로 선언
      let userName = prompt("이름 입력");
      let userAge = prompt("나이 입력");
      callBack(userName,userAge);//callback? showdata써야 할꺼같은데
      }
      getData(showData);//매서드 호출
      이해가 안되니 다시 정리해보자
      getData(x) 이 인자로 함수를 받을 수 있음. 지금은 x가 함수라고 할 경우.
      그리고 그 인자를 받은 함수를 실행할 수 도 있음.
      왜냐하면 getData 실행 시 x매서드는 showData이기 때문임.
      따라서 매서드 실행 후 지역변수에 값을 입력 후 콜백함수(인자로 받았던 함수)를 호출
      따라서 showData가 실행되는 것임.
      만약, showData() 괄호를 붙여서 인자로 준다면
      showData의 return값을 주는 데 해당 매서드의 return이 없다면 undefined를 반환하게 됨

      시간을 고려하는 타이머 함수

      일정 시간 마다 반복 - setInterval(콜백함스,시간)
      반복 실행 멈추기 - clearInterval(타이머)
      let counter = 0;
      let timer = setInterval(()=>{ //콜백 함수, 타이머 시작
      console.log("안녕하세요?");
      counter++;
      if(counter === 5){//counter가 5가되면 해당 함수 실행
      clearInterval(timer);//setInterval로 시작한 함수 종료
      }
      },2000)//2000밀리초 뒤에 종료 2초
      특정 시간 이후에 실행 - setTimeout(콜백함수, 시간)
      setTimeout(()=>{
      console.log("안녕하세요?");
      },3000)//3초뒤 실행
