var와 변수 호이스팅
var x = 10;
var sum = x+y;
var y = 20;
console.log(sum);//값은 NaN 을 출력
차라리 let이나 const라면 오류를 발생시키겠지만 var라서 오류 없음.
var는 지양해야 함.

변수 선언 후 재할당
var라면 변수를 선언한 뒤 재할당 재선언 가능
var sum = add(10,20);
var sum = 100; //이런식으로 재선언과 재할당
let을 재할당만 가능 let=100;이런식으로

자료형

숫자나 문자열 등 처리해야 할 자료의 유형 = 자료형
원시 유형 과 객체로 분리

typeof(값 또는 변수)로 확인해보기
typeof(data);//"number" 변수가 숫자형이란 것을 암시, typeof(10);//'number', typeof("10");// 'string'
특수 기호도 \r (캐리지리턴) 등도 사용 가능
console.log('탭 \t 포함');
탬플릿 리터럴을 사용하여 + 기호를 사용하지 않고 ``와 ${변수명}을 통해 문자열에서 쉽게 변수출력 가능
let name="백두산";
let classroom = 201;
console.log(`${name}님, ${classroom}호 강의실로 입장하세요.`);
논리형 은 true,false , truthy, falsy
nudefined 은 값할당 x , null은 값이 없거나 유효하지 않음

객체 정의
let gitBook = { title : "깃&깃허브 입문", pubDate : "2019-12-06", pages : 272, finished : true }
에서 gitBook.title , gitBook["finished"] 로 값 가져올 수 있음
배열
emptyArr=[]
colors = ["red","blue","green"];//혹은 여러 자료형 도 가능
colors[2]; 를 통해 배열 내의 값 확인 가능

Symbol() 유일성을 보장
let var1 = Symbol();
let var2 = Symbol();
var1==var2//false를 반환 , 한번 만들면 변경 불가능
활용은 이렇게 가능
let id = Symbol();
const member = { name : "kim", [id] : 12345, };
member // name과 symbol() 으로 출력됨
member[id] // 12345를출력
member.id = 6789 라고 다시 지정하면 심벌 과 id로 따로 표현됨
