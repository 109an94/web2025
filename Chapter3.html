증감연산자 활용 방법 : let x = 10, let y = 4;
result = x+y-- //result : 10, y:3이 됨.
result = ++x-y // y:3 result:8 x:11
연결연산자 : + (다른 자료형과 합쳐 문자열로 만들 때 사용)
할당연산자 : += ... %=
비교연사자 : ==, != 등 (=== !==는 값과 데이터 유형까지 같아야 함)
따라서 3 =="3" //true를 반환하지만 3 === "3"//false를 반환
문자열끼리 비교 "a">"b" //false 가 나옴 왜? ASCII코드 값을 바탕으로 비교하기 때문이다.

switch문 : switch() case#: break;
let subject = prompt("신청할 과목을 선택하세요. 1.HTML 2.CSS 3.JavaScript")
if(subject !== null){
switch(subject){
case "1": document.write("HTML을 신청했습니다.");
break;
case "2": document.write("CSS를 신청했습니다.");
break;
case "3": document.write("JavaScript을 신청했습니다.");
break;
default : document.write("잘못 입력했습니다. 다시 입력해 주세요.")
}
}
